    23	  pub fn load(data_root: &Path) -> Self {
    24	    let dir = data_root.join("credentials");
    25	    if let Err(err) = std::fs::create_dir_all(&dir) {
    26	      warn!("Couldn't create {}: {err}", dir.display());
    27	    }
    28	    let (path, legacy) = (dir.join(MIDJOURNEY_FILENAME), vec![dir.join(MIDJOURNEY_LEGACY_FILENAME)]);
    29	    let midjourney = MidjourneyCredentialManager::load(path.clone(), legacy.clone()).unwrap_or_else(|err| {
    30	      warn!("Couldn't load the Midjourney login: {err}");

    32	    });
    33	    Self { midjourney, fal_key_path: Arc::new(dir.join(FAL_KEY_FILENAME)) }
    34	  }

   146	/// Starts `fields` (the image page's OmniGen request) at `provider`; `image_urls` are the

   148	pub async fn submit(host: &ApiHost, creds: Option<&StorytellerCredentialSet>, accounts: &ProviderAccounts, provider: Provider, fields: &Map<String, Value>, image_urls: Vec<String>) -> Result<ProviderJob, SubmitError> {
   149	  match provider {
   150	    Provider::Midjourney => submit_midjourney(host, creds, accounts, fields, image_urls).await,
   151	    Provider::Fal => submit_fal(host, creds, accounts, fields, image_urls).await,

   156	async fn submit_midjourney(host: &ApiHost, creds: Option<&StorytellerCredentialSet>, accounts: &ProviderAccounts, fields: &Map<String, Value>, image_urls: Vec<String>) -> Result<ProviderJob, SubmitError> {
   157	  let model_id = model_id(fields);
   158	  if !is_midjourney(model_id) {
   159	    return Err("This model cannot use a Midjourney account".into());
   160	  }
   161	  if !image_urls.is_empty() {
   162	    return Err("Direct Midjourney does not support image references yet".into());
   163	  }
   164	  if fields.get("image_batch_count").and_then(Value::as_u64).is_some_and(|n| n != u64::from(MIDJOURNEY_BATCH)) {
   165	    return Err("Direct Midjourney generates four images per job".into());
   166	  }
   167	  let mut builder = request_builder(RouterProvider::Midjourney, fields, Vec::new())?;
   168	  builder.resolution = None;
   169	  builder.quality = None;
   170	  builder.image_batch_count = Some(MIDJOURNEY_BATCH);
   171	  let prompt_request = router_image_request_to_artcraft_prompt(&builder);
   172	  let Ok(ImageGenerationDraftOrRequest::Request(generation)) = builder.build2() else {
   173	    return Err("Unexpected Midjourney request draft".into());
   174	  };
   175	  let session = accounts.midjourney.session().await.map_err(|err| err.to_string())?.ok_or(SubmitError::NeedsAccount(Provider::Midjourney))?;
   176	  // The prompt comes first, so the results can be attributed to it after a restart.
   177	  let creds = creds.ok_or("Please sign in to generate.")?;
   178	  let prompt = create_prompt(host, Some(creds), prompt_request).await.map_err(|err| api_message(&err))?;
   179	  let client = RouterClient::Midjourney(RouterMidjourneyClient::new(session.cookie_header, session.user_id, session.browser));
   180	  let GenerateImageResponse::Midjourney(response) = generation.send_request(&client).await.map_err(router_message(Provider::Midjourney))? else {
   181	    return Err("Unexpected Midjourney response".into());
   182	  };
   183	  Ok(new_job(Provider::Midjourney, fields, u32::from(MIDJOURNEY_BATCH), response.job_id, None, None, Some(prompt.prompt_token.to_string())))
   184	}

   186	async fn submit_fal(host: &ApiHost, creds: Option<&StorytellerCredentialSet>, accounts: &ProviderAccounts, fields: &Map<String, Value>, image_urls: Vec<String>) -> Result<ProviderJob, SubmitError> {
   187	  let key = accounts.fal_key().ok_or(SubmitError::NeedsAccount(Provider::Fal))?;
   188	  let builder = request_builder(RouterProvider::Fal, fields, image_urls)?;
   189	  // The prompt record is a nicety here: the generation goes ahead without it.
   190	  let maybe_prompt_token = match creds {
   191	    Some(creds) => create_prompt(host, Some(creds), router_image_request_to_artcraft_prompt(&builder)).await.map(|p| p.prompt_token.to_string()).map_err(|err| warn!("Couldn't create the prompt record: {}", api_message(&err))).ok(),
   192	    None => None,
   193	  };
   194	  let generation = match builder.build2() {
   195	    Ok(ImageGenerationDraftOrRequest::Request(generation)) => generation,
   196	    Ok(ImageGenerationDraftOrRequest::Draft(_)) => return Err("Fal should not be sending draft requests".into()),
   197	    Err(err) => return Err(err.to_string().into()),
   198	  };
   199	  let client = RouterClient::Fal(RouterFalClient::new_polling_only_from_raw_key(&key));
   200	  let GenerateImageResponse::Fal(response) = generation.send_request(&client).await.map_err(router_message(Provider::Fal))? else {
   201	    return Err("Unexpected FAL response".into());
   202	  };
   203	  let request_id = response.request_id.or(response.gateway_request_id).ok_or("FAL didn't return a request id")?;
   204	  let count = fields.get("image_batch_count").and_then(Value::as_u64).map_or(1, |n| n.max(1) as u32);
   205	  Ok(new_job(Provider::Fal, fields, count, request_id, response.maybe_status_url, response.maybe_response_url, maybe_prompt_token))
   206	}

   208	/// The router's request for the page's fields. The setting values are the router's own names
   209	/// (`wide_sixteen_by_nine`, `one_k`, `high`).
   210	fn request_builder(provider: RouterProvider, fields: &Map<String, Value>, image_urls: Vec<String>) -> Result<GenerateImageRequestBuilder, String> {
   211	  let model = match (parse::<RouterImageModel>(fields.get("model")), model_id(fields)) {
   212	    (Some(model), _) => model,
   213	    // The generic Midjourney model is the newest version.
   214	    (None, "midjourney") => RouterImageModel::Midjourney8,
   215	    (None, id) => return Err(format!("{id} isn't available on this provider")),
   216	  };
   217	  let mismatch = if matches!(provider, RouterProvider::Midjourney) { GenerationModeMismatchStrategy::AbortGeneration } else { GenerationModeMismatchStrategy::GenerateAnyway };
   218	  Ok(GenerateImageRequestBuilder { model, provider, prompt: fields.get("prompt").and_then(Value::as_str).map(str::to_owned), image_inputs: (!image_urls.is_empty()).then_some(ImageListRef::Urls(image_urls)), resolution: parse(fields.get("resolution")), aspect_ratio: parse(fields.get("aspect_ratio")), quality: parse(fields.get("quality")), image_batch_count: fields.get("image_batch_count").and_then(Value::as_u64).map(|n| n as u16), horizontal_angle: None, vertical_angle: None, zoom: None, request_mismatch_mitigation_strategy: RequestMismatchMitigationStrategy::PayMoreUpgrade, generation_mode_mismatch_strategy: Some(mismatch), idempotency_token: None })
   219	}

   343	  /// What the model can do on this provider (`imageModelForProvider`): direct Midjourney makes
   344	  /// four images, without references, resolution or quality choices.
   345	  pub fn adapt(self, model: &ModelInfo) -> ModelInfo {
   346	    let mut m = model.clone();
   347	    if self == Provider::Midjourney && is_midjourney(&model.id) {
   348	      m.batch_sizes = vec![MIDJOURNEY_BATCH];
   349	      m.batch_default = MIDJOURNEY_BATCH;
   350	      m.image_refs_max = 0;
   351	      m.resolutions.clear();
   352	      m.resolution_default = None;
   353	      m.qualities.clear();
   354	      m.quality_default = None;
   355	    }
   356	    m
   357	  }
