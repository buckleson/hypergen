// Guards against the old product name returning to user-facing sources. Only
// the billing and generative-AI files below may still mention it, because they
// talk to the upstream account service. Reads the files as text; runs nothing.

import { existsSync, readdirSync, readFileSync, statSync } from "node:fs";
import { dirname, join } from "node:path";
import { fileURLToPath } from "node:url";

import { describe, expect, it } from "vitest";

const root = join(dirname(fileURLToPath(import.meta.url)), "..", "..", "..");

const OLD_NAME = /diffusion|Diffusion Studio/i;
const CONTRACT_IMPORT = "@diffusionstudio/api-contract";

const SCANNED_FILES = [
  "apps/web/index.html",
  "apps/desktop/src/main.ts",
  "apps/desktop/src/tray.ts",
  "apps/desktop/src/menu.ts",
  "apps/desktop/src/mcp-config.ts",
  "apps/desktop/src/mcp-install.ts",
];

const SCANNED_DIRS = ["apps/web/src/components", "apps/cli/src", "packages/agent-chat/src/host", "docs"];

const ALLOWED = [
  /^apps\/web\/src\/components\/upgrade-dialog\.tsx$/,
  /^apps\/web\/src\/components\/purchase-success\.tsx$/,
  /^apps\/web\/src\/components\/genai\//,
  /^apps\/web\/src\/components\/dashboard\/(account|ai-credits|billing-info|plans|invoices)-view\.tsx$/,
];

function walk(dir: string): string[] {
  return readdirSync(join(root, dir)).flatMap((name) => {
    const path = `${dir}/${name}`;
    if (name === "node_modules") return [];
    return statSync(join(root, path)).isDirectory() ? walk(path) : [path];
  });
}

function isAllowed(path: string): boolean {
  return ALLOWED.some((pattern) => pattern.test(path));
}

/** Lines of `text` naming the old product, ignoring imports of the upstream API contract. */
function matchingLines(text: string): string[] {
  return text
    .split(/\r?\n/)
    .filter((line) => !(/\bimport\b|\bfrom\b/.test(line) && line.includes(CONTRACT_IMPORT)))
    .filter((line) => OLD_NAME.test(line));
}

function offendingLines(path: string): string[] {
  return matchingLines(readFileSync(join(root, path), "utf8"));
}

function offenders(paths: string[], read: (path: string) => string[] = offendingLines): string[] {
  return paths
    .filter((path) => !isAllowed(path))
    .flatMap((path) => read(path).map((line) => `${path}: ${line.trim()}`));
}

const files = [...SCANNED_FILES, ...SCANNED_DIRS.flatMap(walk)];

describe("branding guard", () => {
  it("scans a non-trivial set of files", () => {
    expect(files.length).toBeGreaterThan(100);
  });

  it("keeps the old product name out of every file outside the allow-list", () => {
    expect(offenders(files)).toEqual([]);
  });
});

describe("branding guard: scanned set", () => {
  it("every explicitly listed file exists", () => {
    for (const path of SCANNED_FILES) expect(existsSync(join(root, path)), path).toBe(true);
  });

  it("every scanned directory contributes at least one file", () => {
    for (const dir of SCANNED_DIRS) {
      expect(files.some((path) => path.startsWith(`${dir}/`)), dir).toBe(true);
    }
  });

  it("uses forward-slash repo-relative paths on every platform (no Windows backslashes)", () => {
    for (const path of files) {
      expect(path.includes("\\")).toBe(false);
      expect(path).not.toMatch(/^[A-Za-z]:/);
      expect(path.startsWith("/")).toBe(false);
    }
  });

  it("skips node_modules and lists each file once", () => {
    expect(files.some((path) => path.split("/").includes("node_modules"))).toBe(false);
    expect(new Set(files).size).toBe(files.length);
  });

  it("covers React components and the CLI sources", () => {
    expect(files).toContain("apps/web/src/components/upgrade-dialog.tsx");
    expect(files.some((path) => path.startsWith("apps/cli/src/") && path.endsWith(".ts"))).toBe(true);
  });
});

describe("branding guard: line matcher", () => {
  it("returns nothing for empty or clean text", () => {
    expect(matchingLines("")).toEqual([]);
    expect(matchingLines("\n\r\n\n")).toEqual([]);
    expect(matchingLines("Hypergen\nWelcome to Hypergen")).toEqual([]);
  });

  it.each([
    "Diffusion Studio",
    "DIFFUSION STUDIO",
    "diffusion",
    "<title>diffusion studio</title>",
    'icon="diffusion-logo"',
    "https://app.diffusion.studio/callback",
    "diffusion://open",
    "const DIFFUSION_ANALYTICS = 1;",
    "StableDiffusion",
  ])("flags %j case-insensitively", (line) => {
    expect(matchingLines(line)).toEqual([line]);
  });

  it("ignores import lines of the api contract", () => {
    expect(matchingLines('import { Plan } from "@diffusionstudio/api-contract";')).toEqual([]);
    expect(matchingLines('import type { Plan } from "@diffusionstudio/api-contract/plans";')).toEqual([]);
    expect(matchingLines('} from "@diffusionstudio/api-contract";')).toEqual([]);
    expect(matchingLines('import "@diffusionstudio/api-contract";')).toEqual([]);
  });

  it("still flags api-contract mentions that are not import lines", () => {
    const line = 'const pkg = "@diffusionstudio/api-contract";';
    expect(matchingLines(line)).toEqual([line]);
  });

  it("still flags imports of other old-brand packages", () => {
    const line = 'import x from "@diffusionstudio/core";';
    expect(matchingLines(line)).toEqual([line]);
  });

  it("does not let an ignored import hide a match on another line", () => {
    const text = 'import { A } from "@diffusionstudio/api-contract";\nexport const label = "Diffusion Studio";';
    expect(matchingLines(text)).toEqual(['export const label = "Diffusion Studio";']);
  });

  it("splits CRLF and LF line endings without leaking carriage returns", () => {
    expect(matchingLines("ok\r\nDiffusion Studio\r\nok")).toEqual(["Diffusion Studio"]);
    expect(matchingLines("a\ndiffusion\nb\ndiffusion")).toEqual(["diffusion", "diffusion"]);
  });
});

describe("branding guard: failure reporting", () => {
  const fake: Record<string, string> = {
    "apps/cli/src/zz-probe.ts": 'export const name = "Diffusion Studio";',
    "apps/web/src/components/upgrade-dialog.tsx": "Upgrade your Diffusion Studio plan",
    "apps/web/src/components/genai/x.tsx": "diffusion",
    "apps/web/src/components/dashboard/billing-view.tsx": "  Billed by diffusion  ",
    "docs/clean.md": "Hypergen docs",
  };
  const read = (path: string) => matchingLines(fake[path] ?? "");

  it("fails on a non-allowed file and names the file and trimmed line", () => {
    expect(offenders(Object.keys(fake), read)).toEqual([
      'apps/cli/src/zz-probe.ts: export const name = "Diffusion Studio";',
      "apps/web/src/components/dashboard/billing-view.tsx: Billed by diffusion",
    ]);
  });

  it("passes when only allowed files match", () => {
    expect(
      offenders(["apps/web/src/components/upgrade-dialog.tsx", "apps/web/src/components/genai/x.tsx"], read),
    ).toEqual([]);
  });

  it("passes on an empty file set", () => {
    expect(offenders([], read)).toEqual([]);
  });
});

describe("branding guard: no third-party brand leaks", () => {
  it("no scanned user-facing file mentions LTX or Lightricks", () => {
    const leaks = files.filter((path) => /\bLTX\b|Lightricks/i.test(readFileSync(join(root, path), "utf8")));
    expect(leaks).toEqual([]);
  });
});

// Brand strings in the main process and in the repo-root files that ship with
// it (README, .gitignore, Windows CI). main.ts starts Electron on import, so
// these checks read the sources as text instead of executing them.

import { readFileSync } from "node:fs";
import { join } from "node:path";
import { describe, expect, it } from "vitest";

const src = __dirname;
const root = join(src, "..", "..", "..");
const read = (...p: string[]) => readFileSync(join(...p), "utf8");
const lines = (text: string) => text.split(/\r?\n/);

// Lines that may still say "dapi": the DapiServer import/identifier, the
// `dapi` variable and the internal @hypergen/dapi package import.
const allowedDapi = (line: string) =>
  /DapiServer|\bdapi\.|const dapi\b|@hypergen\/dapi|\.\/dapi\//.test(line);

const leaks = (text: string) =>
  lines(text).filter((l) => /diffusion|LTX|Lightricks|\bdapi\b/i.test(l) && !allowedDapi(l));

describe("main.ts", () => {
  const main = read(src, "main.ts");

  it("names the app Hypergen", () => {
    expect(main).toMatch(/app\.setName\("Hypergen"\)/);
    expect(main.match(/app\.setName\(/g)).toHaveLength(1);
  });

  it("checks for updates against buckleson/hypergen", () => {
    expect(main).toMatch(/repo:\s*"buckleson\/hypergen"/);
  });

  it("uses the hypergen deep-link scheme", () => {
    expect(main).toMatch(/const AUTH_PROTOCOL = "hypergen";/);
    expect(main).toContain("hypergen://");
    expect(main).toMatch(/hypergen CLI/);
  });

  it("has no upstream brand outside the allowed dapi identifiers", () => {
    expect(leaks(main)).toEqual([]);
  });
});

describe("squirrel.ts", () => {
  const squirrel = read(src, "squirrel.ts");

  it("comment names only the hypergen command", () => {
    expect(squirrel).toContain("`hypergen`");
    expect(leaks(squirrel)).toEqual([]);
  });
});

describe("squirrel.test.ts fixtures", () => {
  const t = read(src, "squirrel.test.ts");

  it("use Hypergen install paths and links", () => {
    expect(t).toContain("\\\\Local\\\\Hypergen\\\\app-1.0.0\\\\Hypergen.exe");
    expect(t).toContain("hypergen://auth?x=1");
    expect(t).toContain("/Applications/Hypergen.app/Contents/MacOS/Hypergen");
    expect(leaks(t)).toEqual([]);
  });
});

describe("README.md", () => {
  const readme = read(root, "README.md");

  it("describes Hypergen and clones from buckleson/hypergen", () => {
    expect(readme).toMatch(/Hypergen/);
    expect(readme).toContain("git clone https://github.com/buckleson/hypergen.git");

    expect(fenced).not.toMatch(/^\s*\$?\s*dapi\b/m);

  it("has no upstream brand", () => {
    expect(wf).not.toMatch(/diffusion|\bdapi\b|LTX|Lightricks/i);
  });

// Static checks that the usage-tracking, sign-in and checkout plumbing stays out of
// the desktop main process, and that the provider channels are wired once.

import { existsSync, readdirSync, readFileSync, statSync } from "node:fs";
import { join } from "node:path";
import { describe, expect, it } from "vitest";
import { MAIN_CHANNELS } from "./main-channels";

const src = __dirname;
// Built from fragments so this file never matches the repo-wide grep it mirrors.
const TRACK_WORD = "analy" + "tics";
const BILLING_WORD = "cred" + "its";
const API_CONTRACT = "api-" + "contract";
const read = (file: string) => readFileSync(join(src, file), "utf8");

function sourceFiles(dir: string): string[] {
  return readdirSync(dir).flatMap((name) => {
    const path = join(dir, name);
    if (statSync(path).isDirectory()) return sourceFiles(path);
    return /\.(ts|tsx)$/.test(name) ? [path] : [];
  });
}

describe("main process removals", () => {
  it("has no usage-tracking module", () => {
    expect(existsSync(join(src, `${TRACK_WORD}.ts`))).toBe(false);
  });

  it("main.ts has no tracking, protocol client or deep-link routing", () => {
    const main = read("main.ts");
    for (const banned of [
      new RegExp(TRACK_WORD, "i"),
      /AUTH_PROTOCOL/,
      /setAsDefaultProtocolClient/,
      /pendingDeepLinks/,
      /deliverDeepLink/,
      /takePendingDeepLink/,
      /deepLinkChannel/,
      /["']open-url["']/,
      new RegExp(`${TRACK_WORD.toUpperCase()}_CONFIGURE`),
      /\bAUTH_[A-Z_]+/,
      /\bCHECKOUT_[A-Z_]+/,
    ]) {
      expect(main, String(banned)).not.toMatch(banned);
    }
  });

  it("second-instance still focuses the window", () => {
    const main = read("main.ts");
    const handler = main.slice(main.indexOf('"second-instance"'));
    expect(main).toContain('"second-instance"');
    expect(handler.slice(0, 300)).toMatch(/windows\.show\(\)/);
  });

  it("main.ts registers the provider handlers exactly once, with keys.json under userData", () => {
    const main = read("main.ts");
    const calls = main.match(/registerProviderHandlers\(/g) ?? [];
    expect(calls).toHaveLength(1);
    expect(main).toMatch(/registerProviderHandlers\(join\(app\.getPath\("userData"\), "keys\.json"\)\)/);
  });

  it("main-channels drops tracking, auth, checkout and deep links", () => {
    const names = Object.keys(MAIN_CHANNELS);
    expect(names.filter((n) => new RegExp(`^(${TRACK_WORD.toUpperCase()}_|AUTH_|CHECKOUT_)`).test(n))).toEqual([]);
    const channels = read("main-channels.ts");
    expect(channels).not.toMatch(/DeepLinkChannel/);
    expect(channels).not.toMatch(new RegExp(`\\b(${TRACK_WORD.toUpperCase()}_CONFIGURE|AUTH_[A-Z_]+|CHECKOUT_[A-Z_]+)\\b`));
  });

  it("main-channels has the eight provider channels with unique wire names", () => {
    const expected = [
      "PROVIDERS_KEYS_STATUS",
      "PROVIDERS_KEYS_SET",
      "PROVIDERS_KEYS_REMOVE",
      "PROVIDERS_KEYS_TEST",
      "PROVIDERS_FAL_SUBMIT",
      "PROVIDERS_FAL_POLL",
      "PROVIDERS_FAL_CANCEL",
      "PROVIDERS_GEMINI_GENERATE",
    ];
    const channels = MAIN_CHANNELS as Record<string, string>;
    for (const name of expected) expect(typeof channels[name], name).toBe("string");
    const values = Object.values(channels);
    expect(new Set(values).size).toBe(values.length);
    const source = read("main-channels.ts");
    for (const name of expected) expect(source).toMatch(new RegExp(`\\[MAIN_CHANNELS\\.${name}\\]:\\s*\\{`));
  });

  it("dapi/server.ts has no tool-call tracking", () => {
    const server = read(join("dapi", "server.ts"));
    expect(server).not.toMatch(/TrackProps/);
    expect(server).not.toMatch(/onToolCall/);
  });

  it("projects.ts no longer counts code changes", () => {
    expect(read("projects.ts")).not.toMatch(/countCodeChange/);
  });

  it("no desktop source mentions the banned billing/tracking words", () => {
    const banned = new RegExp(`${TRACK_WORD}|${API_CONTRACT}|${BILLING_WORD}`, "i");
    const hits = sourceFiles(src).filter((file) => banned.test(readFileSync(file, "utf8")));
    expect(hits).toEqual([]);
  });

  it("the provider modules carry no legacy branding or telemetry", () => {
    const dir = join(src, "providers");
    for (const file of readdirSync(dir).filter((n) => !n.includes(".test."))) {
      const text = readFileSync(join(dir, file), "utf8");
      expect(text, file).not.toMatch(/\b(LTX|Lightricks|sentry|posthog|telemetry)\b/i);
    }
  });
});

// Lines that may still say dapi/diffusion: package imports, the api-contract
// import, Dapi*/DAPI_* identifiers, and the dapiBinary export.
function leftovers(path: string, pattern: RegExp): string[] {
  return read(path)
    .split("\n")
    .filter((line) => pattern.test(line))
    .filter((line) => !/from "@hypergen\/dapi"|from "@diffusionstudio\/api-contract"/.test(line))
    .filter((line) => !/^\s*\/\/.*@hypergen\/dapi/.test(line))
    .filter((line) => pattern.test(line.replace(/\bDapi\w*|\bDAPI_\w+|\bdapiBinary\b|\bisDapiError\b|@hypergen\/dapi|apps\/web\/src\/dapi/g, "")));
}
